        private void RegistrarPaciente()
        {
            if (!ValidarDatos())
                return;

            if (pacientes.Any(p => p.Dni == txtDni.Text.Trim()))
            {
                MessageBox.Show("Ya existe un paciente con ese DNI.");
                return;
            }

            pacientes.Add(ObtenerDatosFormulario());

            MessageBox.Show("Paciente registrado correctamente.");
            LimpiarFormulario();
        }

        private void ActualizarPaciente()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= pacientes.Count)
            {
                MessageBox.Show("Seleccione un paciente de la tabla.");
                return;
            }

            if (!ValidarDatos())
                return;

            string dni = txtDni.Text.Trim();

            if (pacientes.Any(p =>
                p.Dni == dni && p != pacientes[indiceSeleccionado]))
            {
                MessageBox.Show("El DNI ya pertenece a otro paciente.");
                return;
            }

            PacienteVista actualizado = ObtenerDatosFormulario();
            PacienteVista anterior = pacientes[indiceSeleccionado];

            anterior.Dni = actualizado.Dni;
            anterior.Nombres = actualizado.Nombres;
            anterior.Apellidos = actualizado.Apellidos;
            anterior.FechaNacimiento = actualizado.FechaNacimiento;
            anterior.Telefono = actualizado.Telefono;
            anterior.Direccion = actualizado.Direccion;

            dgvPacientes.Refresh();

            MessageBox.Show("Paciente actualizado correctamente.");
            LimpiarFormulario();
        }

        private void EliminarPaciente()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= pacientes.Count)
            {
                MessageBox.Show("Seleccione un paciente de la tabla.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Está seguro de eliminar este paciente?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Question);

            if (respuesta == DialogResult.Yes)
            {
                pacientes.RemoveAt(indiceSeleccionado);
                MessageBox.Show("Paciente eliminado.");
                LimpiarFormulario();
            }
        }

        private void SeleccionarPaciente()
        {
            if (dgvPacientes.CurrentRow == null)
                return;

            if (dgvPacientes.CurrentRow.DataBoundItem
                is not PacienteVista paciente)
                return;

            indiceSeleccionado = pacientes.IndexOf(paciente);

            txtDni.Text = paciente.Dni;
            txtNombres.Text = paciente.Nombres;
            txtApellidos.Text = paciente.Apellidos;
            dtpNacimiento.Value = paciente.FechaNacimiento;
            txtTelefono.Text = paciente.Telefono;
            txtDireccion.Text = paciente.Direccion;
        }

        private bool ValidarDatos()
        {
            if (string.IsNullOrWhiteSpace(txtDni.Text) ||
                string.IsNullOrWhiteSpace(txtNombres.Text) ||
                string.IsNullOrWhiteSpace(txtApellidos.Text))
            {
                MessageBox.Show(
                    "Complete el DNI, los nombres y los apellidos.");
                return false;
            }

            if (txtDni.Text.Trim().Length != 8 ||
                !txtDni.Text.Trim().All(char.IsDigit))
            {
                MessageBox.Show(
                    "El DNI debe contener exactamente 8 números.");
                return false;
            }

            return true;
        }

        private PacienteVista ObtenerDatosFormulario()
        {
            return new PacienteVista
            {
                Dni = txtDni.Text.Trim(),
                Nombres = txtNombres.Text.Trim(),
                Apellidos = txtApellidos.Text.Trim(),
                FechaNacimiento = dtpNacimiento.Value.Date,
                Telefono = txtTelefono.Text.Trim(),
                Direccion = txtDireccion.Text.Trim()
            };
        }

        private void LimpiarFormulario()
        {
            indiceSeleccionado = -1;

            txtDni.Clear();
            txtNombres.Clear();
            txtApellidos.Clear();
            txtTelefono.Clear();
            txtDireccion.Clear();

            dtpNacimiento.Value = DateTime.Today;
            dgvPacientes.ClearSelection();
        }

        private class PacienteVista
        {
            public string Dni { get; set; } = "";
            public string Nombres { get; set; } = "";
            public string Apellidos { get; set; } = "";
            public DateTime FechaNacimiento { get; set; }
            public string Telefono { get; set; } = "";
            public string Direccion { get; set; } = "";
        }
    }

        private static System.Collections.Generic.IEnumerable<T>
            TodosLosControles<T>(Control raiz) where T : Control
        {
            foreach (Control control in raiz.Controls)
            {
                if (control is T tipado)
                    yield return tipado;

                foreach (T hijo in TodosLosControles<T>(control))
                    yield return hijo;
            }
        }

private void AplicarEstilosRecursivos(Control control)
{
    if (control is GroupBox grupo)
    {
        TemaSIGEPOSTA.EstilizarGrupo(grupo);
    }
    else if (control is TextBox caja)
    {
        if (caja.Multiline)
        {
            caja.BackColor = TemaSIGEPOSTA.FondoCampo;
            caja.ForeColor = TemaSIGEPOSTA.Texto;
            caja.Font = TemaSIGEPOSTA.FuenteCampo;
            caja.BorderStyle = BorderStyle.FixedSingle;
        }
        else
        {
            TemaSIGEPOSTA.EstilizarTextBox(caja);
        }
    }
    else if (control is ComboBox combo)
    {
        TemaSIGEPOSTA.EstilizarComboBox(combo);
    }
    else if (control is Label etiqueta)
    {
        TemaSIGEPOSTA.EstilizarEtiqueta(etiqueta);
    }
    else if (control is Button boton)
    {
        TipoBoton tipo = boton.Text.Contains("ELIMINAR")
            ? TipoBoton.Peligro
            : boton.Text.Contains("LIMPIAR")
                ? TipoBoton.Secundario
                : TipoBoton.Primario;

        TemaSIGEPOSTA.EstilizarBoton(boton, tipo);
    }

    foreach (Control hijo in control.Controls)
    {
        AplicarEstilosRecursivos(hijo);
    }
}

        private void Registrar()
        {
            if (!Validar())
                return;

            diagnosticos.Add(ObtenerDatos());

            MessageBox.Show("Diagnóstico registrado correctamente.");
            Limpiar();
        }

        private void Actualizar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= diagnosticos.Count)
            {
                MessageBox.Show("Seleccione un diagnóstico.");
                return;
            }

            if (!Validar())
                return;

            DiagnosticoVista nuevo = ObtenerDatos();
            DiagnosticoVista anterior = diagnosticos[indiceSeleccionado];

            anterior.DniPaciente = nuevo.DniPaciente;
            anterior.Paciente = nuevo.Paciente;
            anterior.DniProfesional = nuevo.DniProfesional;
            anterior.Profesional = nuevo.Profesional;
            anterior.Codigo = nuevo.Codigo;
            anterior.Fecha = nuevo.Fecha;
            anterior.Tipo = nuevo.Tipo;
            anterior.Estado = nuevo.Estado;
            anterior.Descripcion = nuevo.Descripcion;
            anterior.Observaciones = nuevo.Observaciones;

            dgvDiagnosticos.Refresh();

            MessageBox.Show("Diagnóstico actualizado correctamente.");
            Limpiar();
        }

        private void Eliminar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= diagnosticos.Count)
            {
                MessageBox.Show("Seleccione un diagnóstico.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Desea eliminar este diagnóstico?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning
            );

            if (respuesta == DialogResult.Yes)
            {
                diagnosticos.RemoveAt(indiceSeleccionado);

                MessageBox.Show("Diagnóstico eliminado.");
                Limpiar();
            }
        }

        private void Seleccionar()
        {
            if (dgvDiagnosticos.CurrentRow?.DataBoundItem
                is not DiagnosticoVista d)
                return;

            indiceSeleccionado = diagnosticos.IndexOf(d);

            txtDniPaciente.Text = d.DniPaciente;
            txtPaciente.Text = d.Paciente;
            txtDniProfesional.Text = d.DniProfesional;
            txtProfesional.Text = d.Profesional;
            txtCodigo.Text = d.Codigo;
            dtpFecha.Value = d.Fecha;
            cmbTipo.SelectedItem = d.Tipo;
            cmbEstado.SelectedItem = d.Estado;
            txtDescripcion.Text = d.Descripcion;
            txtObservaciones.Text = d.Observaciones;
        }

        private bool Validar()
        {
            if (string.IsNullOrWhiteSpace(txtDniPaciente.Text) ||
                string.IsNullOrWhiteSpace(txtPaciente.Text) ||
                string.IsNullOrWhiteSpace(txtDniProfesional.Text) ||
                string.IsNullOrWhiteSpace(txtProfesional.Text) ||
                string.IsNullOrWhiteSpace(txtCodigo.Text) ||
                string.IsNullOrWhiteSpace(txtDescripcion.Text))
            {
                MessageBox.Show("Complete los campos obligatorios.");
                return false;
            }

            if (!DniValido(txtDniPaciente.Text) ||
                !DniValido(txtDniProfesional.Text))
            {
                MessageBox.Show(

            }

        }

        private bool DniValido(string dni)

private void AplicarEstilosRecursivos(Control control)
{
    if (control is GroupBox grupo)
    {
        TemaSIGEPOSTA.EstilizarGrupo(grupo);
    }
    else if (control is TextBox caja)
    {
        if (caja.Multiline)
        {
            caja.BackColor = TemaSIGEPOSTA.FondoCampo;
            caja.ForeColor = TemaSIGEPOSTA.Texto;
            caja.Font = TemaSIGEPOSTA.FuenteCampo;
            caja.BorderStyle = BorderStyle.FixedSingle;
        }
        else
        {
            TemaSIGEPOSTA.EstilizarTextBox(caja);
        }
    }
    else if (control is ComboBox combo)
    {
        TemaSIGEPOSTA.EstilizarComboBox(combo);
    }
    else if (control is Label etiqueta)
    {
        TemaSIGEPOSTA.EstilizarEtiqueta(etiqueta);
    }
    else if (control is Button boton)
    {
        TipoBoton tipo = boton.Text.Contains("ELIMINAR")
            ? TipoBoton.Peligro
            : boton.Text.Contains("LIMPIAR")
                ? TipoBoton.Secundario
                : TipoBoton.Primario;

        TemaSIGEPOSTA.EstilizarBoton(boton, tipo);
    }

    foreach (Control hijo in control.Controls)
    {
        AplicarEstilosRecursivos(hijo);
    }
}

        private void Registrar()
        {
            if (!Validar())
                return;

            if (medicamentos.Any(m =>
                m.Codigo.Equals(
                    txtCodigo.Text.Trim(),
                    StringComparison.OrdinalIgnoreCase)))
            {
                MessageBox.Show(
                    "Ya existe un medicamento con ese código.");
                return;
            }

            medicamentos.Add(ObtenerDatos());

            MessageBox.Show("Medicamento registrado correctamente.");
            Limpiar();
        }

        private void Actualizar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= medicamentos.Count)
            {
                MessageBox.Show("Seleccione un medicamento.");
                return;
            }

            if (!Validar())
                return;

            string codigo = txtCodigo.Text.Trim();

            if (medicamentos.Where((m, i) => i != indiceSeleccionado)
                .Any(m => m.Codigo.Equals(
                    codigo,
                    StringComparison.OrdinalIgnoreCase)))
            {
                MessageBox.Show(
                    "El código ya pertenece a otro medicamento.");
                return;
            }

            MedicamentoVista nuevo = ObtenerDatos();
            MedicamentoVista anterior = medicamentos[indiceSeleccionado];

            anterior.Codigo = nuevo.Codigo;
            anterior.Nombre = nuevo.Nombre;
            anterior.PrincipioActivo = nuevo.PrincipioActivo;
            anterior.Concentracion = nuevo.Concentracion;
            anterior.Presentacion = nuevo.Presentacion;
            anterior.Laboratorio = nuevo.Laboratorio;
            anterior.Categoria = nuevo.Categoria;
            anterior.Stock = nuevo.Stock;
            anterior.StockMinimo = nuevo.StockMinimo;
            anterior.Precio = nuevo.Precio;
            anterior.Vencimiento = nuevo.Vencimiento;
            anterior.Estado = nuevo.Estado;
            anterior.Observacion = nuevo.Observacion;

            dgvMedicamentos.Refresh();

            MessageBox.Show("Medicamento actualizado correctamente.");
            Limpiar();
        }

        private void Eliminar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= medicamentos.Count)
            {
                MessageBox.Show("Seleccione un medicamento.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Desea eliminar este medicamento del catálogo?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning
            );

            if (respuesta == DialogResult.Yes)
            {
                medicamentos.RemoveAt(indiceSeleccionado);

                MessageBox.Show("Medicamento eliminado.");
                Limpiar();
            }
        }

        private void Seleccionar()
        {
            if (dgvMedicamentos.CurrentRow?.DataBoundItem
                is not MedicamentoVista m)
                return;

            indiceSeleccionado = medicamentos.IndexOf(m);

            txtCodigo.Text = m.Codigo;
            txtNombre.Text = m.Nombre;
            txtPrincipioActivo.Text = m.PrincipioActivo;
            txtConcentracion.Text = m.Concentracion;
            txtPresentacion.Text = m.Presentacion;
            txtLaboratorio.Text = m.Laboratorio;

            cmbCategoria.SelectedItem = m.Categoria;

            txtStock.Text = m.Stock.ToString();
            txtStockMinimo.Text = m.StockMinimo.ToString();
            txtPrecio.Text = m.Precio.ToString("0.00");

            dtpVencimiento.Value = m.Vencimiento;
            cmbEstado.SelectedItem = m.Estado;
            txtObservacion.Text = m.Observacion ?? string.Empty;
        }

        private bool Validar()
        {
            if (string.IsNullOrWhiteSpace(txtCodigo.Text) ||
                string.IsNullOrWhiteSpace(txtNombre.Text) ||
                string.IsNullOrWhiteSpace(txtPrincipioActivo.Text) ||
                string.IsNullOrWhiteSpace(txtConcentracion.Text) ||
                string.IsNullOrWhiteSpace(txtPresentacion.Text))
            {
                MessageBox.Show("Complete los campos obligatorios.");
                return false;
            }

            if (!int.TryParse(txtStock.Text, out int stock) || stock < 0)
            {
                MessageBox.Show("Ingrese un stock válido.");
                return false;
            }

            if (!int.TryParse(txtStockMinimo.Text, out int minimo) ||
                minimo < 0)
            {
                MessageBox.Show("Ingrese un stock mínimo válido.");
                return false;
            }

            if (!decimal.TryParse(txtPrecio.Text, out decimal precio) ||
                precio < 0)
            {
                MessageBox.Show("Ingrese un precio válido.");
                return false;
            }

            if (cmbCategoria.SelectedItem == null ||
                cmbEstado.SelectedItem == null)
            {
                MessageBox.Show(
                    "Seleccione una categoría y un estado.");
                return false;
            }

            return true;
        }

        private MedicamentoVista ObtenerDatos()
        {
            return new MedicamentoVista
            {
                Codigo = txtCodigo.Text.Trim(),
                Nombre = txtNombre.Text.Trim(),
                PrincipioActivo = txtPrincipioActivo.Text.Trim(),
                Concentracion = txtConcentracion.Text.Trim(),
                Presentacion = txtPresentacion.Text.Trim(),
                Laboratorio = txtLaboratorio.Text.Trim(),
                Categoria = cmbCategoria.Text,
                Stock = int.Parse(txtStock.Text),
                StockMinimo = int.Parse(txtStockMinimo.Text),
                Precio = decimal.Parse(txtPrecio.Text),
                Vencimiento = dtpVencimiento.Value.Date,
                Estado = cmbEstado.Text,
                Observacion = txtObservacion.Text.Trim()
            };
        }

        private void Limpiar()
        {
            indiceSeleccionado = -1;

            txtCodigo.Clear();
            txtNombre.Clear();
            txtPrincipioActivo.Clear();
            txtConcentracion.Clear();
            txtPresentacion.Clear();
            txtLaboratorio.Clear();
            txtStock.Clear();
            txtStockMinimo.Clear();
            txtPrecio.Clear();
            txtObservacion.Clear();

            cmbCategoria.SelectedIndex = 0;
            cmbEstado.SelectedIndex = 0;
            dtpVencimiento.Value = DateTime.Today;

            dgvMedicamentos.ClearSelection();
        }

private void AplicarEstilos()
{
    BackColor = TemaSIGEPOSTA.Fondo;
    Font = TemaSIGEPOSTA.Fuente;

    AplicarEstilosRecursivos(this);

    // Fecha de registro
    TemaSIGEPOSTA.EstilizarFecha(dtpFechaRegistro);

    // Tabla de historias clínicas
    TemaSIGEPOSTA.EstilizarTabla(dgvHistorias);
}

private void AplicarEstilosRecursivos(Control control)
{
    if (control is GroupBox grupo)
    {
        TemaSIGEPOSTA.EstilizarGrupo(grupo);
    }
    else if (control is TextBox caja)
    {
        if (caja.Multiline)
        {
            caja.BackColor = TemaSIGEPOSTA.FondoCampo;
            caja.ForeColor = TemaSIGEPOSTA.Texto;
            caja.Font = TemaSIGEPOSTA.FuenteCampo;
            caja.BorderStyle = BorderStyle.FixedSingle;
        }
        else
        {
            TemaSIGEPOSTA.EstilizarTextBox(caja);
        }
    }
    else if (control is Label etiqueta)
    {
        TemaSIGEPOSTA.EstilizarEtiqueta(etiqueta);
    }
    else if (control is Button boton)
    {
        TipoBoton tipo = boton.Text.Contains("ELIMINAR")
            ? TipoBoton.Peligro
            : boton.Text.Contains("LIMPIAR")
                ? TipoBoton.Secundario
                : TipoBoton.Primario;

        TemaSIGEPOSTA.EstilizarBoton(boton, tipo);
    }

    foreach (Control hijo in control.Controls)
    {
        AplicarEstilosRecursivos(hijo);
    }
}

        private void Registrar()
        {
            if (!Validar())
                return;

            if (historias.Any(h => h.Dni == txtDni.Text.Trim()))
            {
                MessageBox.Show(
                    "Este paciente ya tiene una historia clínica registrada.");

                return;
            }

            historias.Add(ObtenerDatos());

            MessageBox.Show("Historia clínica registrada correctamente.");
            Limpiar();
        }

        private void Actualizar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= historias.Count)
            {
                MessageBox.Show("Seleccione una historia clínica.");
                return;
            }

            if (!Validar())
                return;

            string dni = txtDni.Text.Trim();

            if (historias.Any(h =>
                h.Dni == dni &&
                h != historias[indiceSeleccionado]))
            {
                MessageBox.Show(
                    "El DNI pertenece a otra historia clínica.");

                return;
            }

            HistoriaVista nuevo = ObtenerDatos();
            HistoriaVista anterior = historias[indiceSeleccionado];

            anterior.Dni = nuevo.Dni;
            anterior.Paciente = nuevo.Paciente;
            anterior.GrupoSanguineo = nuevo.GrupoSanguineo;
            anterior.Alergias = nuevo.Alergias;
            anterior.AntecedentesPersonales = nuevo.AntecedentesPersonales;
            anterior.AntecedentesFamiliares = nuevo.AntecedentesFamiliares;
            anterior.Observaciones = nuevo.Observaciones;
            anterior.FechaRegistro = nuevo.FechaRegistro;

            dgvHistorias.Refresh();

            MessageBox.Show("Historia clínica actualizada correctamente.");
            Limpiar();
        }

        private void Eliminar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= historias.Count)
            {
                MessageBox.Show("Seleccione una historia clínica.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Está seguro de eliminar esta historia clínica?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning
            );

            if (respuesta == DialogResult.Yes)
            {
                historias.RemoveAt(indiceSeleccionado);

                MessageBox.Show("Registro eliminado.");
                Limpiar();
            }
        }

        private void Seleccionar()
        {
            if (dgvHistorias.CurrentRow?.DataBoundItem
                is not HistoriaVista h)
                return;

            indiceSeleccionado = historias.IndexOf(h);

            txtDni.Text = h.Dni;
            txtPaciente.Text = h.Paciente;
            txtGrupoSanguineo.Text = h.GrupoSanguineo;
            txtAlergias.Text = h.Alergias;
            txtAntecedentesPersonales.Text = h.AntecedentesPersonales;
            txtAntecedentesFamiliares.Text = h.AntecedentesFamiliares;
            txtObservaciones.Text = h.Observaciones;
            dtpFechaRegistro.Value = h.FechaRegistro;
        }

        private bool Validar()
        {
            if (string.IsNullOrWhiteSpace(txtDni.Text) ||
                string.IsNullOrWhiteSpace(txtPaciente.Text))
            {
                MessageBox.Show(
                    "Ingrese el DNI y el nombre del paciente.");

                return false;
            }

            string dni = txtDni.Text.Trim();

            if (dni.Length != 8 || !dni.All(char.IsDigit))
            {
                MessageBox.Show(
                    "El DNI debe contener exactamente 8 números.");

                return false;
            }

            return true;
        }

        private HistoriaVista ObtenerDatos()
        {
            return new HistoriaVista
            {
                Dni = txtDni.Text.Trim(),
                Paciente = txtPaciente.Text.Trim(),
                GrupoSanguineo = txtGrupoSanguineo.Text.Trim(),
                Alergias = txtAlergias.Text.Trim(),
                AntecedentesPersonales = txtAntecedentesPersonales.Text.Trim(),
                AntecedentesFamiliares = txtAntecedentesFamiliares.Text.Trim(),
                Observaciones = txtObservaciones.Text.Trim(),
                FechaRegistro = dtpFechaRegistro.Value.Date
            };
        }

        private void Limpiar()
        {
            indiceSeleccionado = -1;

            txtDni.Clear();
            txtPaciente.Clear();
            txtGrupoSanguineo.Clear();
            txtAlergias.Clear();
            txtAntecedentesPersonales.Clear();
            txtAntecedentesFamiliares.Clear();
            txtObservaciones.Clear();

            dtpFechaRegistro.Value = DateTime.Today;
            dgvHistorias.ClearSelection();
        }

        // ═══════════════════════ MODELO ═══════════════════════

        private class HistoriaVista
        {
            public string Dni { get; set; } = "";
            public string Paciente { get; set; } = "";
            public string GrupoSanguineo { get; set; } = "";
            public string Alergias { get; set; } = "";
            public string AntecedentesPersonales { get; set; } = "";
            public string AntecedentesFamiliares { get; set; } = "";
            public string Observaciones { get; set; } = "";
            public DateTime FechaRegistro { get; set; }
        }
    }

        private void Consultar()
        {
            if (dtpDesde.Value.Date > dtpHasta.Value.Date)
            {
                MessageBox.Show(
                    "La fecha inicial no puede ser posterior a la fecha final.");
                return;
            }

            dgvReporte.DataSource = null;

            lblResumen.Text = "Reporte seleccionado: " + cmbReporte.Text +
                              " | Periodo: " +
                              dtpDesde.Value.ToShortDateString() + " - " +
                              dtpHasta.Value.ToShortDateString();

            MessageBox.Show(
                "La estructura del reporte está preparada. " +
                "Los resultados se conectarán con SQL Server posteriormente."
            );
        }

        private void ExportarCSV()
        {
            if (dgvReporte.Columns.Count == 0 ||
                dgvReporte.Rows.Count == 0)
            {
                MessageBox.Show("No existen resultados para exportar.");
                return;
            }

            using SaveFileDialog guardar = new SaveFileDialog
            {
                Filter = "Archivo CSV (*.csv)|*.csv",
                FileName = "Reporte_SIGEPOSTA.csv",
                Title = "Guardar reporte"
            };

            if (guardar.ShowDialog() != DialogResult.OK)
                return;

            try
            {
                using System.IO.StreamWriter escritor =
                    new System.IO.StreamWriter(
                        guardar.FileName,
                        false,
                        System.Text.Encoding.UTF8);

                string encabezados = string.Join(";",
                    dgvReporte.Columns.Cast<DataGridViewColumn>()
                    .Select(c => EscaparCSV(c.HeaderText)));

                escritor.WriteLine(encabezados);

                foreach (DataGridViewRow fila in dgvReporte.Rows)
                {
                    if (fila.IsNewRow)
                        continue;

                    string valores = string.Join(";",
                        fila.Cells.Cast<DataGridViewCell>()
                        .Select(c => EscaparCSV(c.Value?.ToString() ?? "")));

                    escritor.WriteLine(valores);
                }

                MessageBox.Show("Reporte exportado correctamente.");
            }
            catch (Exception ex)
            {
                MessageBox.Show("Error al exportar: " + ex.Message);
            }
        }

        private string EscaparCSV(string valor)
        {
            return "\"" + valor.Replace("\"", "\"\"") + "\"";
        }

        private void Limpiar()
        {
            cmbReporte.SelectedIndex = 0;
            dtpDesde.Value = DateTime.Today;
            dtpHasta.Value = DateTime.Today;
            dgvReporte.DataSource = null;
            lblResumen.Text = "Sin información para mostrar.";
        }
    }

        private void CrearRespaldo()
        {
            string origen = txtCarpetaOrigen.Text;
            string destino = txtCarpetaRespaldos.Text;

            if (!Directory.Exists(origen))
            {
                MessageBox.Show("Seleccione una carpeta de origen válida.");
                return;
            }

            if (string.IsNullOrWhiteSpace(destino))
            {
                MessageBox.Show("Seleccione la carpeta de respaldos.");
                return;
            }

            string rutaOrigen = Path.GetFullPath(origen)
                .TrimEnd(Path.DirectorySeparatorChar) +
                Path.DirectorySeparatorChar;

            string rutaDestino = Path.GetFullPath(destino)
                .TrimEnd(Path.DirectorySeparatorChar) +
                Path.DirectorySeparatorChar;

            if (rutaDestino.StartsWith(
                rutaOrigen,
                StringComparison.OrdinalIgnoreCase))
            {
                MessageBox.Show(
                    "La carpeta de respaldos no puede estar dentro de la carpeta de origen."
                );
                return;
            }

            try
            {
                Directory.CreateDirectory(destino);

                string nombre = "SIGEPOSTA_Respaldo_" +
                                DateTime.Now.ToString("yyyyMMdd_HHmmss") +
                                ".zip";

                string archivo = Path.Combine(destino, nombre);

                ZipFile.CreateFromDirectory(
                    origen,
                    archivo,
                    CompressionLevel.Optimal,
                    false
                );

                lblEstado.Text = "Estado: respaldo creado correctamente.";

                MessageBox.Show(
                    "Copia de seguridad creada correctamente.\n\n" +
                    archivo
                );

                ListarRespaldos();
            }
            catch (Exception ex)
            {
                lblEstado.Text = "Estado: error al crear el respaldo.";

                MessageBox.Show(
                    "No se pudo crear el respaldo:\n" + ex.Message
                );
            }
        }

        private void ListarRespaldos()
        {
            string carpeta = txtCarpetaRespaldos.Text;

            if (!Directory.Exists(carpeta))
            {
                dgvRespaldos.DataSource = null;
                return;
            }

            var archivos = Directory.GetFiles(carpeta, "*.zip")
                .Select(ruta => new
                {
                    Nombre = Path.GetFileName(ruta),
                    Ubicacion = ruta,
                    Fecha = File.GetLastWriteTime(ruta),
                    Tamano = Math.Round(
                        new FileInfo(ruta).Length / 1024.0,
                        2
                    ) + " KB"
                })
                .OrderByDescending(a => a.Fecha)
                .ToList();

            dgvRespaldos.DataSource = archivos;

            lblEstado.Text = "Estado: " + archivos.Count +
                             " respaldo(s) encontrado(s).";
        }

        private void RestaurarRespaldo()
        {
            if (dgvRespaldos.CurrentRow == null)
            {
                MessageBox.Show("Seleccione un respaldo de la lista.");
                return;
            }

            string archivo = dgvRespaldos.CurrentRow
                .Cells["Ubicacion"].Value?.ToString() ?? "";

            if (!File.Exists(archivo))
            {
                MessageBox.Show("El archivo seleccionado no existe.");
                return;
            }

            using FolderBrowserDialog dialogo = new FolderBrowserDialog
            {
                Description = "Seleccione la carpeta donde se restaurarán los archivos"
            };

            if (dialogo.ShowDialog() != DialogResult.OK)
                return;

            DialogResult respuesta = MessageBox.Show(
                "La restauración puede sobrescribir archivos existentes.\n" +
                "Se recomienda crear un respaldo de la carpeta de destino antes de continuar.\n\n" +
                "¿Desea continuar?",
                "Confirmar restauración",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning
            );

            if (respuesta != DialogResult.Yes)
                return;

            try
            {
                ZipFile.ExtractToDirectory(
                    archivo,
                    dialogo.SelectedPath,
                    true
                );

                lblEstado.Text = "Estado: restauración completada.";

                MessageBox.Show("Respaldo restaurado correctamente.");
            }
            catch (Exception ex)
            {
                lblEstado.Text = "Estado: error al restaurar.";

                MessageBox.Show(
                    "No se pudo restaurar el respaldo:\n" + ex.Message
                );
            }
        }

        private void RegistrarUsuario()
        {
            if (!ValidarCampos())
                return;

            foreach (DataGridViewRow fila in tabla.Rows)
            {
                if (string.Equals(
                    Convert.ToString(fila.Cells["Usuario"].Value),
                    txtUsuario.Text.Trim(),
                    StringComparison.OrdinalIgnoreCase))
                {
                    MessageBox.Show("El nombre de usuario ya está registrado.");
                    return;
                }
            }

            tabla.Rows.Add(
                txtNombre.Text.Trim(),
                txtUsuario.Text.Trim(),
                cmbRol.Text,
                chkActivo.Checked ? "Activo" : "Inactivo"
            );

            MessageBox.Show(
                "Usuario agregado temporalmente.",
                "SIGEPOSTA",
                MessageBoxButtons.OK,
                MessageBoxIcon.Information
            );

            LimpiarCampos();
        }

        private void ActualizarUsuario()
        {
            if (tabla.SelectedRows.Count == 0)
            {
                MessageBox.Show("Seleccione un usuario de la tabla.");
                return;
            }

            if (!ValidarCampos())
                return;

            DataGridViewRow fila = tabla.SelectedRows[0];
            string usuarioNuevo = txtUsuario.Text.Trim();

            foreach (DataGridViewRow otraFila in tabla.Rows)
            {
                if (otraFila == fila)
                    continue;

                if (string.Equals(
                    Convert.ToString(otraFila.Cells["Usuario"].Value),
                    usuarioNuevo,
                    StringComparison.OrdinalIgnoreCase))
                {
                    MessageBox.Show("Ese nombre de usuario ya pertenece a otra cuenta.");
                    return;
                }
            }

            fila.Cells["Nombre"].Value = txtNombre.Text.Trim();
            fila.Cells["Usuario"].Value = usuarioNuevo;
            fila.Cells["Rol"].Value = cmbRol.Text;
            fila.Cells["Estado"].Value = chkActivo.Checked ? "Activo" : "Inactivo";

            MessageBox.Show("Usuario actualizado temporalmente.");
            LimpiarCampos();
        }

        private void EliminarUsuario()
        {
            if (tabla.SelectedRows.Count == 0)
            {
                MessageBox.Show("Seleccione un usuario de la tabla.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Está seguro de eliminar este usuario?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning
            );

            if (respuesta == DialogResult.Yes)
            {
                tabla.Rows.RemoveAt(tabla.SelectedRows[0].Index);
                LimpiarCampos();
            }
        }

        private void SeleccionarUsuario()
        {
            if (tabla.SelectedRows.Count == 0)
                return;

            DataGridViewRow fila = tabla.SelectedRows[0];

            txtNombre.Text = Convert.ToString(fila.Cells["Nombre"].Value) ?? "";
            txtUsuario.Text = Convert.ToString(fila.Cells["Usuario"].Value) ?? "";
            cmbRol.Text = Convert.ToString(fila.Cells["Rol"].Value) ?? "";
            chkActivo.Checked =
                Convert.ToString(fila.Cells["Estado"].Value) == "Activo";

            txtContrasena.Clear();
        }

        private bool ValidarCampos()
        {
            if (string.IsNullOrWhiteSpace(txtNombre.Text) ||
                string.IsNullOrWhiteSpace(txtUsuario.Text) ||
                string.IsNullOrWhiteSpace(txtContrasena.Text))
            {
                MessageBox.Show("Complete el nombre, usuario y contraseña.");
                return false;
            }

            if (txtContrasena.Text.Trim().Length < 8)
            {
                MessageBox.Show("La contraseña debe tener al menos 8 caracteres.");
                return false;
            }

            if (cmbRol.SelectedIndex < 0)
            {
                MessageBox.Show("Seleccione un rol.");
                return false;
            }

            return true;
        }

        private void LimpiarCampos()
        {
            txtNombre.Clear();
            txtUsuario.Clear();
            txtContrasena.Clear();
            cmbRol.SelectedIndex = 1;
            chkActivo.Checked = true;
            tabla.ClearSelection();
            txtNombre.Focus();
        }
    }
