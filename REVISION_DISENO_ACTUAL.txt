    internal static class AtencionEstilo
    {
        public static readonly Color Verde = Color.FromArgb(43, 125, 91);
        public static readonly Color VerdeHover = Color.FromArgb(34, 106, 77);
        public static readonly Color VerdeOscuro = Color.FromArgb(27, 65, 54);
        public static readonly Color Fondo = Color.FromArgb(240, 244, 242);
        public static readonly Color Campo = Color.FromArgb(245, 248, 246);
        public static readonly Color Texto = Color.FromArgb(28, 40, 35);
        public static readonly Color Borde = Color.FromArgb(215, 225, 220);
        public static readonly Color Rojo = Color.FromArgb(200, 55, 55);

        public static GraphicsPath Redondear(Rectangle r, int radio)
        {
            GraphicsPath path = new GraphicsPath();

            if (r.Width <= 0 || r.Height <= 0)
                return path;

            int d = Math.Min(radio * 2, Math.Min(r.Width, r.Height));

            path.AddArc(r.X, r.Y, d, d, 180, 90);
            path.AddArc(r.Right - d, r.Y, d, d, 270, 90);
            path.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90);
            path.AddArc(r.X, r.Bottom - d, d, d, 90, 90);
            path.CloseFigure();

            return path;
        }
    }

    internal class CampoAtencion : UserControl
    {
        private readonly TextBox entrada = new();

        public override string Text
        {
            get => entrada.Text;
            set => entrada.Text = value;
        }

        public bool Multiline => entrada.Multiline;

        public CampoAtencion()
        {
            Height = 40;
            MinimumSize = new Size(80, 40);
            BackColor = AtencionEstilo.Campo;
            Padding = new Padding(10, 5, 10, 5);
            Font = new Font("Segoe UI", 10);
            DoubleBuffered = true;

            SetStyle(ControlStyles.ResizeRedraw, true);

            entrada.BorderStyle = BorderStyle.None;
            entrada.Dock = DockStyle.Fill;
            entrada.BackColor = AtencionEstilo.Campo;
            entrada.ForeColor = AtencionEstilo.Texto;
            entrada.Font = new Font("Segoe UI", 10);
            entrada.Margin = new Padding(0);
            entrada.Multiline = false;

            Controls.Add(entrada);

            entrada.GotFocus += (s, e) => Invalidate();
            entrada.LostFocus += (s, e) => Invalidate();
            entrada.TextChanged += (s, e) => OnTextChanged(EventArgs.Empty);
        }

        public void ConfigurarMultilinea(bool multilinea, ScrollBars barras)
        {
            entrada.Multiline = multilinea;
            entrada.ScrollBars = barras;
            entrada.AcceptsReturn = multilinea;
            entrada.AcceptsTab = multilinea;

            Padding = multilinea
                ? new Padding(10, 9, 10, 9)
                : new Padding(10, 5, 10, 5);

            Invalidate();
        }

        public new void Clear()
        {
            entrada.Clear();
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);

            e.Graphics.SmoothingMode = SmoothingMode.AntiAlias;

            Rectangle r = new Rectangle(1, 1, Width - 3, Height - 3);

            using GraphicsPath path = AtencionEstilo.Redondear(r, 9);

            Color colorBorde = entrada.Focused
                ? AtencionEstilo.Verde
                : AtencionEstilo.Borde;

            float grosor = entrada.Focused ? 1.8f : 1f;

            using SolidBrush fondo = new SolidBrush(AtencionEstilo.Campo);
            using Pen borde = new Pen(colorBorde, grosor);

            e.Graphics.FillPath(fondo, path);
            e.Graphics.DrawPath(borde, path);
        }
    }

    internal class BotonAtencion : Button
    {
        private readonly TipoBoton tipo;
        private bool sobre;
        private bool presionado;

        public BotonAtencion(string texto, TipoBoton tipo)
        {
            this.tipo = tipo;

            Text = texto;
            Size = new Size(155, 46);
            MinimumSize = new Size(125, 44);
            FlatStyle = FlatStyle.Flat;
            FlatAppearance.BorderSize = 0;
            Font = new Font("Segoe UI Semibold", 9.5f);
            Cursor = Cursors.Hand;
            ForeColor = Color.White;
            BackColor = AtencionEstilo.Verde;
            UseVisualStyleBackColor = false;
            Margin = new Padding(5, 4, 5, 4);

            SetStyle(
                ControlStyles.UserPaint |
                ControlStyles.AllPaintingInWmPaint |
                ControlStyles.OptimizedDoubleBuffer,
                true);

            MouseEnter += (s, e) =>
            {
                sobre = true;
                Invalidate();
            };

            MouseLeave += (s, e) =>
            {
                sobre = false;
                presionado = false;
                Invalidate();
            };

            MouseDown += (s, e) =>
            {
                if (e.Button == MouseButtons.Left)
                {
                    presionado = true;
                    Invalidate();
                }
            };

            MouseUp += (s, e) =>
            {
                presionado = false;
                Invalidate();
            };
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            e.Graphics.SmoothingMode = SmoothingMode.AntiAlias;

            Rectangle sombra = new Rectangle(1, 3, Width - 3, Height - 4);
            Rectangle r = new Rectangle(1, 1, Width - 3, Height - 5);

            using GraphicsPath pathSombra = AtencionEstilo.Redondear(sombra, 10);
            using GraphicsPath path = AtencionEstilo.Redondear(r, 10);

            using SolidBrush colorSombra =
                new SolidBrush(Color.FromArgb(28, 0, 0, 0));

            e.Graphics.FillPath(colorSombra, pathSombra);

            Color baseColor;
            Color hoverColor;
            Color textoColor;

            if (tipo == TipoBoton.Peligro)
            {
                baseColor = AtencionEstilo.Rojo;
                hoverColor = Color.FromArgb(165, 40, 40);
                textoColor = Color.White;
            }
            else if (tipo == TipoBoton.Secundario)
            {
                baseColor = Color.White;
                hoverColor = Color.FromArgb(232, 241, 236);
                textoColor = AtencionEstilo.VerdeOscuro;
            }
            else
            {
                baseColor = AtencionEstilo.Verde;
                hoverColor = AtencionEstilo.VerdeHover;
                textoColor = Color.White;
            }

            if (sobre)
                baseColor = hoverColor;

            if (presionado)
            {
                baseColor = tipo == TipoBoton.Secundario
                    ? Color.FromArgb(220, 234, 226)
                    : AtencionEstilo.VerdeOscuro;
            }

            using SolidBrush fondo = new SolidBrush(baseColor);
            e.Graphics.FillPath(fondo, path);

            if (tipo == TipoBoton.Secundario)
            {
                using Pen borde = new Pen(AtencionEstilo.Borde, 1);
                e.Graphics.DrawPath(borde, path);
            }

            TextRenderer.DrawText(
                e.Graphics,
                Text,
                Font,
                r,
                textoColor,
                TextFormatFlags.HorizontalCenter |
                TextFormatFlags.VerticalCenter |
                TextFormatFlags.EndEllipsis);
        }
    }

                ControlStyles.ResizeRedraw,
                true);
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);

            e.Graphics.SmoothingMode = SmoothingMode.AntiAlias;

            Rectangle sombra = new Rectangle(3, 4, Width - 7, Height - 7);
            Rectangle r = new Rectangle(1, 1, Width - 4, Height - 5);

            using GraphicsPath sombraPath = AtencionEstilo.Redondear(sombra, 15);
            using GraphicsPath path = AtencionEstilo.Redondear(r, 15);

            using SolidBrush sombraColor =
                new SolidBrush(Color.FromArgb(17, 30, 50, 40));

            using SolidBrush fondo = new SolidBrush(Color.White);
            using Pen borde = new Pen(AtencionEstilo.Borde, 1);

            e.Graphics.FillPath(sombraColor, sombraPath);
            e.Graphics.FillPath(fondo, path);
            e.Graphics.DrawPath(borde, path);
        }
    }

        private void Registrar()
        {
            if (!Validar())
                return;

            atenciones.Add(ObtenerDatos());

            MessageBox.Show("Atención médica registrada correctamente.");
            Limpiar();
        }

        private void Actualizar()
        {
            if (indiceSeleccionado < 0 || indiceSeleccionado >= atenciones.Count)
            {
                MessageBox.Show("Seleccione una atención del listado.");
                return;
            }

            if (!Validar())
                return;

            AtencionVista nuevo = ObtenerDatos();
            AtencionVista anterior = atenciones[indiceSeleccionado];

            anterior.DniPaciente = nuevo.DniPaciente;
            anterior.Paciente = nuevo.Paciente;
            anterior.DniProfesional = nuevo.DniProfesional;
            anterior.Profesional = nuevo.Profesional;
            anterior.Fecha = nuevo.Fecha;
            anterior.Temperatura = nuevo.Temperatura;
            anterior.Presion = nuevo.Presion;
            anterior.Peso = nuevo.Peso;
            anterior.Talla = nuevo.Talla;
            anterior.Sintomas = nuevo.Sintomas;
            anterior.Evaluacion = nuevo.Evaluacion;
            anterior.Tratamiento = nuevo.Tratamiento;
            anterior.Observaciones = nuevo.Observaciones;
            anterior.Estado = nuevo.Estado;

            dgvAtenciones.Refresh();

            MessageBox.Show("Atención actualizada correctamente.");
            Limpiar();
        }

        private void Eliminar()
        {
            if (indiceSeleccionado < 0 || indiceSeleccionado >= atenciones.Count)
            {
                MessageBox.Show("Seleccione una atención.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Desea eliminar este registro de atención?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Question
            );

            if (respuesta == DialogResult.Yes)
            {
                atenciones.RemoveAt(indiceSeleccionado);
                MessageBox.Show("Registro eliminado.");
                Limpiar();
            }
        }

        private void Seleccionar()
        {
            if (dgvAtenciones.CurrentRow?.DataBoundItem is not AtencionVista a)
                return;

            indiceSeleccionado = atenciones.IndexOf(a);

            txtDniPaciente.Text = a.DniPaciente;
            txtPaciente.Text = a.Paciente;
            txtDniProfesional.Text = a.DniProfesional;
            txtProfesional.Text = a.Profesional;
            dtpFecha.Value = a.Fecha;

            txtTemperatura.Text = a.Temperatura;
            txtPresion.Text = a.Presion;
            txtPeso.Text = a.Peso;
            txtTalla.Text = a.Talla;
            txtSintomas.Text = a.Sintomas;
            txtEvaluacion.Text = a.Evaluacion;
            txtTratamiento.Text = a.Tratamiento;
            txtObservaciones.Text = a.Observaciones;
            cmbEstado.SelectedItem = a.Estado;
        }

        private bool Validar()
        {
            if (string.IsNullOrWhiteSpace(txtDniPaciente.Text) ||
                string.IsNullOrWhiteSpace(txtPaciente.Text) ||
                string.IsNullOrWhiteSpace(txtDniProfesional.Text) ||
                string.IsNullOrWhiteSpace(txtProfesional.Text) ||
                string.IsNullOrWhiteSpace(txtSintomas.Text) ||
                string.IsNullOrWhiteSpace(txtEvaluacion.Text) ||
                string.IsNullOrWhiteSpace(txtTratamiento.Text))
            {
                MessageBox.Show("Complete los datos obligatorios de la atención.");
                return false;
            }

            if (!DniValido(txtDniPaciente.Text) ||
                !DniValido(txtDniProfesional.Text))
            {
                MessageBox.Show("Los DNI deben tener exactamente 8 dígitos.");
                return false;
            }

            if (!NumeroValido(txtTemperatura.Text, 25, 45) ||
                !NumeroValido(txtPeso.Text, 0.1, 500) ||
                !NumeroValido(txtTalla.Text, 0.1, 2.8))
            {
                MessageBox.Show("Revise la temperatura, el peso y la talla.");
                return false;
            }

            if (cmbEstado.SelectedItem == null)
            {
                MessageBox.Show("Seleccione el estado de la atención.");
                return false;
            }

            return true;
        }

        private bool DniValido(string dni)
        {
            return dni.Trim().Length == 8 &&
                   dni.Trim().All(char.IsDigit);
        }

        private bool NumeroValido(
            string valor,
            double minimo,
            double maximo)
        {
            if (string.IsNullOrWhiteSpace(valor))
                return true;

            string normalizado = valor.Trim().Replace(',', '.');

            return double.TryParse(
                normalizado,
                System.Globalization.NumberStyles.AllowDecimalPoint,
                System.Globalization.CultureInfo.InvariantCulture,
                out double numero
            ) && numero >= minimo && numero <= maximo;
        }

        private AtencionVista ObtenerDatos()
        {
            return new AtencionVista
            {
                DniPaciente = txtDniPaciente.Text.Trim(),
                Paciente = txtPaciente.Text.Trim(),
                DniProfesional = txtDniProfesional.Text.Trim(),
                Profesional = txtProfesional.Text.Trim(),
                Fecha = dtpFecha.Value.Date,
                Temperatura = txtTemperatura.Text.Trim(),
                Presion = txtPresion.Text.Trim(),
                Peso = txtPeso.Text.Trim(),
                Talla = txtTalla.Text.Trim(),
                Sintomas = txtSintomas.Text.Trim(),
                Evaluacion = txtEvaluacion.Text.Trim(),
                Tratamiento = txtTratamiento.Text.Trim(),
                Observaciones = txtObservaciones.Text.Trim(),
                Estado = cmbEstado.Text
            };
        }

        private void Limpiar()
        {
            indiceSeleccionado = -1;

            txtDniPaciente.Clear();
            txtPaciente.Clear();
            txtDniProfesional.Clear();
            txtProfesional.Clear();
            txtTemperatura.Clear();
            txtPresion.Clear();
            txtPeso.Clear();
            txtTalla.Clear();
            txtSintomas.Clear();
            txtEvaluacion.Clear();
            txtTratamiento.Clear();
            txtObservaciones.Clear();

            dtpFecha.Value = DateTime.Today;
            cmbEstado.SelectedIndex = 0;

            dgvAtenciones.ClearSelection();
        }

        private class AtencionVista
        {
            public string DniPaciente { get; set; } = "";
            public string Paciente { get; set; } = "";
            public string DniProfesional { get; set; } = "";
            public string Profesional { get; set; } = "";
            public DateTime Fecha { get; set; }
            public string Temperatura { get; set; } = "";
            public string Presion { get; set; } = "";
            public string Peso { get; set; } = "";
            public string Talla { get; set; } = "";
            public string Sintomas { get; set; } = "";
            public string Evaluacion { get; set; } = "";
            public string Tratamiento { get; set; } = "";
            public string Observaciones { get; set; } = "";
            public string Estado { get; set; } = "";
        }

        public static readonly Color Verde = Color.FromArgb(43, 125, 91);
        public static readonly Color VerdeHover = Color.FromArgb(34, 106, 77);
        public static readonly Color VerdeOscuro = Color.FromArgb(27, 65, 54);

        public static readonly Color Borde = Color.FromArgb(215, 225, 220);
        public static readonly Color Rojo = Color.FromArgb(200, 55, 55);

        public static GraphicsPath Redondear(Rectangle r, int radio)
        {
            GraphicsPath p = new();

            if (r.Width <= 0 || r.Height <= 0)
                return p;

            int d = Math.Min(radio * 2, Math.Min(r.Width, r.Height));

            p.AddArc(r.X, r.Y, d, d, 180, 90);
            p.AddArc(r.Right - d, r.Y, d, d, 270, 90);
            p.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90);
            p.AddArc(r.X, r.Bottom - d, d, d, 90, 90);
            p.CloseFigure();

            return p;
        }
    }

    internal class BotonInventario : Button
    {
        private readonly TipoBoton tipo;
        private bool encima;
        private bool presionado;

        public BotonInventario(string texto, TipoBoton tipo)
        {
            Text = texto;
            this.tipo = tipo;

            Size = new Size(155, 46);
            Font = new Font("Segoe UI Semibold", 9.5F);
            Cursor = Cursors.Hand;

            FlatStyle = FlatStyle.Flat;
            FlatAppearance.BorderSize = 0;

            BackColor = Color.Transparent;
            ForeColor = Color.White;
            UseVisualStyleBackColor = false;

            SetStyle(
                ControlStyles.UserPaint |
                ControlStyles.AllPaintingInWmPaint |
                ControlStyles.OptimizedDoubleBuffer |
                ControlStyles.ResizeRedraw,
                true);

            MouseEnter += (s, e) =>
            {
                encima = true;
                Invalidate();
            };

            MouseLeave += (s, e) =>
            {
                encima = false;
                presionado = false;
                Invalidate();
            };

            MouseDown += (s, e) =>
            {
                if (e.Button == MouseButtons.Left)
                {
                    presionado = true;
                    Invalidate();
                }
            };

            MouseUp += (s, e) =>
            {
                presionado = false;
                Invalidate();
            };
        }

        protected override void OnPaintBackground(PaintEventArgs e)
        {
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            Graphics g = e.Graphics;
            g.SmoothingMode = SmoothingMode.AntiAlias;

            g.Clear(Parent?.BackColor ?? InventarioEstilo.Fondo);

            Rectangle sombra = new(2, 4, Width - 5, Height - 5);
            Rectangle r = new(1, 1, Width - 4, Height - 6);

            Color principal;
            Color secundario;

            if (tipo == TipoBoton.Peligro)
            {
                principal = InventarioEstilo.Rojo;
                secundario = Color.FromArgb(165, 43, 43);
            }
            else if (tipo == TipoBoton.Secundario)
            {
                principal = Color.White;
                secundario = Color.FromArgb(225, 240, 229);
            }
            else
            {
                principal = InventarioEstilo.Verde;
                secundario = InventarioEstilo.VerdeHover;
            }

            if (presionado)
            {
                principal = secundario;
            }
            else if (encima)
            {
                Color temporal = principal;
                principal = secundario;
                secundario = temporal;
            }

            using GraphicsPath sombraPath =
                InventarioEstilo.Redondear(sombra, 10);

            using SolidBrush sombraBrush =
                new(Color.FromArgb(30, 0, 0, 0));

            g.FillPath(sombraBrush, sombraPath);

            using GraphicsPath path =
                InventarioEstilo.Redondear(r, 10);

            using LinearGradientBrush degradado = new(
                r, principal, secundario, LinearGradientMode.Vertical);

            g.FillPath(degradado, path);

            Color colorTexto =
                tipo == TipoBoton.Secundario
                    ? InventarioEstilo.VerdeOscuro
                    : Color.White;

            if (!Enabled)
                colorTexto = Color.Gray;

            TextRenderer.DrawText(
                g,
                Text,
                Font,
                r,
                colorTexto,
                TextFormatFlags.HorizontalCenter |
                TextFormatFlags.VerticalCenter);
        }

        // =====================================================
        // REGISTRAR
        // =====================================================

        private void Registrar()
        {
            if (!Validar())
                return;

            MovimientoVista movimiento = ObtenerDatos();

            if (movimiento.Tipo == "Salida" ||
                movimiento.Tipo == "Ajuste negativo")
            {
                MessageBox.Show(
                    "El registro definitivo de salidas y ajustes requiere verificar las existencias del medicamento. Esa validación se integrará con la base de datos."
                );
            }

            movimientos.Add(movimiento);

            MessageBox.Show(
                "Movimiento registrado en el historial temporal.");

            Limpiar();
        }

        // =====================================================
        // ACTUALIZAR
        // =====================================================

        private void Actualizar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= movimientos.Count)
            {
                MessageBox.Show("Seleccione un movimiento.");
                return;
            }

            if (!Validar())
                return;

            MovimientoVista nuevo = ObtenerDatos();
            MovimientoVista anterior = movimientos[indiceSeleccionado];

            anterior.CodigoMedicamento = nuevo.CodigoMedicamento;
            anterior.Medicamento = nuevo.Medicamento;
            anterior.Tipo = nuevo.Tipo;
            anterior.Cantidad = nuevo.Cantidad;
            anterior.Fecha = nuevo.Fecha;
            anterior.Motivo = nuevo.Motivo;
            anterior.Responsable = nuevo.Responsable;
            anterior.Documento = nuevo.Documento;
            anterior.Observacion = nuevo.Observacion;

            dgvMovimientos.Refresh();

            MessageBox.Show("Movimiento actualizado.");

            Limpiar();
        }

        // =====================================================
        // ELIMINAR
        // =====================================================

        private void Eliminar()
        {
            if (indiceSeleccionado < 0 ||
                indiceSeleccionado >= movimientos.Count)
            {
                MessageBox.Show("Seleccione un movimiento.");
                return;
            }

            DialogResult respuesta = MessageBox.Show(
                "¿Desea eliminar este registro del historial temporal?",
                "Confirmar eliminación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning);

            if (respuesta == DialogResult.Yes)
            {
                movimientos.RemoveAt(indiceSeleccionado);

                MessageBox.Show("Registro eliminado.");

                Limpiar();
            }
        }

        // =====================================================
        // SELECCIONAR
        // =====================================================

        private void Seleccionar()
        {
            if (dgvMovimientos.CurrentRow?.DataBoundItem
                is not MovimientoVista m)
                return;

            indiceSeleccionado = movimientos.IndexOf(m);

            txtCodigoMedicamento.Text = m.CodigoMedicamento;
            txtMedicamento.Text = m.Medicamento;
            cmbTipoMovimiento.SelectedItem = m.Tipo;
            txtCantidad.Text = m.Cantidad.ToString();
            dtpFecha.Value = m.Fecha;
            cmbMotivo.SelectedItem = m.Motivo;
            txtResponsable.Text = m.Responsable;
            txtDocumento.Text = m.Documento;
            txtObservacion.Text = m.Observacion;
        }

        // =====================================================
        // VALIDAR
        // =====================================================

        private bool Validar()
        {
            if (string.IsNullOrWhiteSpace(txtCodigoMedicamento.Text) ||
                string.IsNullOrWhiteSpace(txtMedicamento.Text) ||
                string.IsNullOrWhiteSpace(txtResponsable.Text))
            {
                MessageBox.Show("Complete los campos obligatorios.");
                return false;
            }

            if (!int.TryParse(txtCantidad.Text, out int cantidad) ||
                cantidad <= 0)
            {
                MessageBox.Show(
                    "La cantidad debe ser un número mayor que cero.");

                return false;
            }

            if (cmbTipoMovimiento.SelectedItem == null ||
                cmbMotivo.SelectedItem == null)
            {
                MessageBox.Show(
                    "Seleccione el tipo de movimiento y el motivo.");

                return false;
            }

            return true;
        }

        // =====================================================
        // OBTENER DATOS
        // =====================================================

        private MovimientoVista ObtenerDatos()
        {
            return new MovimientoVista
            {
                CodigoMedicamento = txtCodigoMedicamento.Text.Trim(),
                Medicamento = txtMedicamento.Text.Trim(),
                Tipo = cmbTipoMovimiento.Text,
                Cantidad = int.Parse(txtCantidad.Text),
                Fecha = dtpFecha.Value.Date,
                Motivo = cmbMotivo.Text,
                Responsable = txtResponsable.Text.Trim(),
                Documento = txtDocumento.Text.Trim(),
                Observacion = txtObservacion.Text.Trim()
            };
        }

        // =====================================================
        // LIMPIAR
        // =====================================================

        private void Limpiar()
        {
            indiceSeleccionado = -1;

            txtCodigoMedicamento.Clear();
            txtMedicamento.Clear();
            txtCantidad.Clear();
            txtResponsable.Clear();
            txtDocumento.Clear();
            txtObservacion.Clear();

            cmbTipoMovimiento.SelectedIndex = 0;
            cmbMotivo.SelectedIndex = 0;

            dtpFecha.Value = DateTime.Today;

            dgvMovimientos.ClearSelection();
        }

        // =====================================================
        // MODELO
        // =====================================================

        private class MovimientoVista
        {
            public string CodigoMedicamento { get; set; } = "";
            public string Medicamento { get; set; } = "";
            public string Tipo { get; set; } = "";
            public int Cantidad { get; set; }
            public DateTime Fecha { get; set; }
            public string Motivo { get; set; } = "";
            public string Responsable { get; set; } = "";
            public string Documento { get; set; } = "";
            public string Observacion { get; set; } = "";
        }
    }

        private static string ObtenerIniciales(string nombre)
        {
            var partes = nombre.Split(' ', StringSplitOptions.RemoveEmptyEntries);
            if (partes.Length == 0) return "U";
            if (partes.Length == 1) return partes[0][..1].ToUpper();
            return (partes[0][..1] + partes[1][..1]).ToUpper();
        }

        private void ActualizarEncabezado(string titulo)
        {
            lblSeccion.Text = titulo;
            lblRuta.Text = "SIGEPOSTA   ›   " + titulo;

            foreach (var b in botonesMenu)
                b.Activo = (b.Tag as string) == titulo;
        }

        // ------------------------- Panel de inicio -------------------------

        private void MostrarInicio()
        {
            CerrarModuloActual();
            LimpiarPanelTrabajo();
            ActualizarEncabezado("Panel principal");

            var dashboard = new DashboardInicio
            {
                Dock = DockStyle.Fill
            };

            panelTrabajo.Controls.Add(dashboard);
            dashboard.BringToFront();
        }

        // ------------------------- Navegación entre módulos -------------------------

        private void AbrirFormulario<T>(string titulo) where T : Form, new()
        {
            try
            {
                CerrarModuloActual();
                LimpiarPanelTrabajo();
                ActualizarEncabezado(titulo);

                formularioActual = new T
                {
                    TopLevel = false,
                    FormBorderStyle = FormBorderStyle.None,
                    Dock = DockStyle.Fill,
                    BackColor = fondo
                };

                panelTrabajo.Controls.Add(formularioActual);
                formularioActual.Show();
                formularioActual.BringToFront();
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "No se pudo abrir el módulo \"" + titulo + "\".\n\n" + ex.Message,
                    "SIGEPOSTA",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning);
                MostrarInicio();
            }
        }

        private void LimpiarPanelTrabajo()
        {
            foreach (Control c in panelTrabajo.Controls.Cast<Control>().ToList())
            {
                panelTrabajo.Controls.Remove(c);
                c.Dispose();
            }
        }

        private void CerrarModuloActual()
        {
            if (formularioActual != null)
            {
                formularioActual.Close();
                formularioActual.Dispose();
                formularioActual = null;
            }
        }

        private void CerrarSistema()
        {
            DialogResult resultado = MessageBox.Show(
                "¿Está seguro de que desea cerrar SIGEPOSTA?",
                "Cerrar sistema",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Question);

            if (resultado == DialogResult.Yes)
                Close();
        }

        protected override void OnFormClosed(FormClosedEventArgs e)
        {
            CerrarModuloActual();
            base.OnFormClosed(e);
        }
    }

    // ------------------------- Utilidades de dibujo -------------------------

    internal static class Dibujo
    {
        public static GraphicsPath Redondeado(Rectangle r, int radio)
        {
            int d = radio * 2;
            var p = new GraphicsPath();
            p.AddArc(r.X, r.Y, d, d, 180, 90);
            p.AddArc(r.Right - d - 1, r.Y, d, d, 270, 90);
            p.AddArc(r.Right - d - 1, r.Bottom - d - 1, d, d, 0, 90);
            p.AddArc(r.X, r.Bottom - d - 1, d, d, 90, 90);
            p.CloseFigure();
            return p;
        }

        // Cruz médica centrada en (cx, cy)
        public static void CruzMedica(Graphics g, Brush brush, float cx, float cy, float largo, float grosor)
        {
            using var h = Redondeado(new Rectangle(
                (int)(cx - largo / 2), (int)(cy - grosor / 2), (int)largo, (int)grosor), (int)(grosor / 3));
            using var v = Redondeado(new Rectangle(
                (int)(cx - grosor / 2), (int)(cy - largo / 2), (int)grosor, (int)largo), (int)(grosor / 3));
            g.FillPath(brush, h);
            g.FillPath(brush, v);
        }
    }

    {

                ? "Segoe Fluent Icons"

    internal sealed class BotonMenu : Control
    {
        private static readonly Font FuenteIcono = Iconos.Crear(12.5F);
        private static readonly Font FuenteTexto = new("Segoe UI", 9.5F);
        private static readonly Font FuenteActiva = new("Segoe UI", 9.5F, FontStyle.Bold);

        private bool hover;
        private bool activo;

        [Browsable(false)]
        [DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
        public string Icono { get; set; } = "";

        [Browsable(false)]
        [DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
        public bool Activo
        {
            get => activo;
            set { if (activo != value) { activo = value; Invalidate(); } }
        }

        public BotonMenu()
        {
            SetStyle(ControlStyles.UserPaint | ControlStyles.AllPaintingInWmPaint |
                     ControlStyles.OptimizedDoubleBuffer | ControlStyles.ResizeRedraw, true);
            Cursor = Cursors.Hand;
        }

        protected override void OnMouseEnter(EventArgs e) { hover = true; Invalidate(); base.OnMouseEnter(e); }
        protected override void OnMouseLeave(EventArgs e) { hover = false; Invalidate(); base.OnMouseLeave(e); }

        protected override void OnPaint(PaintEventArgs e)
        {
            var g = e.Graphics;
            g.Clear(BackColor);
            g.SmoothingMode = SmoothingMode.AntiAlias;

            var area = new Rectangle(12, 2, Width - 24, Height - 4);

            if (Activo)
            {
                using var path = Dibujo.Redondeado(area, 9);
                using var fondo = new SolidBrush(TemaSIGEPOSTA.Verde);
                g.FillPath(fondo, path);

                using var marca = Dibujo.Redondeado(new Rectangle(area.X + 2, area.Y + 9, 4, area.Height - 18), 2);
                using var blanco = new SolidBrush(Color.White);
                g.FillPath(blanco, marca);
            }
            else if (hover)
            {
                using var path = Dibujo.Redondeado(area, 9);
                using var fondo = new SolidBrush(Color.FromArgb(28, Color.White));
                g.FillPath(fondo, path);
            }

            Color colorTexto = Activo ? Color.White : Color.FromArgb(222, 236, 227);

            TextRenderer.DrawText(g, Icono, FuenteIcono,
                new Rectangle(24, 0, 30, Height), colorTexto,
                TextFormatFlags.HorizontalCenter | TextFormatFlags.VerticalCenter | TextFormatFlags.NoPadding);

            TextRenderer.DrawText(g, Text, Activo ? FuenteActiva : FuenteTexto,
                new Rectangle(62, 0, Width - 74, Height), colorTexto,
                TextFormatFlags.Left | TextFormatFlags.VerticalCenter |
                TextFormatFlags.EndEllipsis | TextFormatFlags.NoPadding);
        }
    }

    internal sealed class TarjetaModulo : Control
    {
        private static readonly Font FuenteIcono = Iconos.Crear(19F);
        private static readonly Font FuenteFlecha = Iconos.Crear(8.5F);
        private static readonly Font FuenteTitulo = new("Segoe UI", 11, FontStyle.Bold);
        private static readonly Font FuenteDetalle = new("Segoe UI", 8.5F);
        private static readonly Font FuentePie = new("Segoe UI", 8, FontStyle.Bold);

        private bool hover;

        [Browsable(false)]
        [DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
        public string Icono { get; set; } = "";
        [Browsable(false)]
        [DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
        public string Titulo { get; set; } = "";
        [Browsable(false)]
        [DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
        public string Descripcion { get; set; } = "";

        public TarjetaModulo()
        {
            SetStyle(ControlStyles.UserPaint | ControlStyles.AllPaintingInWmPaint |
                     ControlStyles.OptimizedDoubleBuffer | ControlStyles.ResizeRedraw, true);
            Cursor = Cursors.Hand;
        }

        protected override void OnMouseEnter(EventArgs e) { hover = true; Invalidate(); base.OnMouseEnter(e); }
        protected override void OnMouseLeave(EventArgs e) { hover = false; Invalidate(); base.OnMouseLeave(e); }

        protected override void OnPaint(PaintEventArgs e)
        {
            var g = e.Graphics;
            g.Clear(BackColor);
            g.SmoothingMode = SmoothingMode.AntiAlias;

            var tarjeta = new Rectangle(3, 2, Width - 9, Height - 10);

            // Sombra suave
            for (int i = 3; i >= 1; i--)
            {
                var sombra = new Rectangle(tarjeta.X - i, tarjeta.Y - i + 3, tarjeta.Width + i * 2, tarjeta.Height + i * 2);
                using var sp = Dibujo.Redondeado(sombra, 14 + i);
                using var sb = new SolidBrush(Color.FromArgb(hover ? 14 : 7, 20, 60, 40));
                g.FillPath(sb, sp);
            }

            // Fondo y borde
            using (var path = Dibujo.Redondeado(tarjeta, 14))
            {
                using var fondo = new SolidBrush(Color.White);
                g.FillPath(fondo, path);
                using var borde = new Pen(hover ? TemaSIGEPOSTA.Verde : Color.FromArgb(225, 232, 228), hover ? 1.6F : 1F);
                g.DrawPath(borde, path);
            }

            int x = tarjeta.X + 20;
            int ancho = tarjeta.Width - 40;

            // Círculo con ícono
            var circulo = new Rectangle(x, tarjeta.Y + 18, 46, 46);
            using (var b = new SolidBrush(hover ? TemaSIGEPOSTA.Verde : TemaSIGEPOSTA.VerdeClaro))
                g.FillEllipse(b, circulo);

            TextRenderer.DrawText(g, Icono, FuenteIcono, circulo,
                hover ? Color.White : TemaSIGEPOSTA.Verde,
                TextFormatFlags.HorizontalCenter | TextFormatFlags.VerticalCenter | TextFormatFlags.NoPadding);

            // Textos
            var flags = TextFormatFlags.Left | TextFormatFlags.VerticalCenter |
                        TextFormatFlags.EndEllipsis | TextFormatFlags.NoPadding;

            TextRenderer.DrawText(g, Titulo, FuenteTitulo,
                new Rectangle(x, tarjeta.Y + 72, ancho, 24), TemaSIGEPOSTA.VerdeOscuro, flags);

            TextRenderer.DrawText(g, Descripcion, FuenteDetalle,
                new Rectangle(x, tarjeta.Y + 96, ancho, 18), TemaSIGEPOSTA.TextoSecundario, flags);

            // Pie: "Abrir módulo ?"
            const string pie = "ABRIR MÓDULO";
            var tam = TextRenderer.MeasureText(g, pie, FuentePie, Size.Empty, TextFormatFlags.NoPadding);
            TextRenderer.DrawText(g, pie, FuentePie,
                new Rectangle(x, tarjeta.Y + 116, tam.Width + 4, 18), TemaSIGEPOSTA.Verde, flags);

            TextRenderer.DrawText(g, "\uE72A", FuenteFlecha,
                new Rectangle(x + tam.Width + 6 + (hover ? 4 : 0), tarjeta.Y + 116, 18, 18),
                TemaSIGEPOSTA.Verde,
                TextFormatFlags.Left | TextFormatFlags.VerticalCenter | TextFormatFlags.NoPadding);
        }
    }
