    public static void EstilizarBoton(
        Button boton,
        TipoBoton tipo = TipoBoton.Primario)
    {
        Color fondo;
        Color hover;
        Color activo;
        Color texto;
        Color borde;

        switch (tipo)
        {
            case TipoBoton.Peligro:
                fondo = Rojo;
                hover = RojoHover;
                activo = RojoActivo;
                texto = Color.White;
                borde = Rojo;
                break;

            case TipoBoton.Secundario:
                fondo = Color.White;
                hover = VerdeClaro;
                activo = Color.FromArgb(205, 228, 212);
                texto = VerdeOscuro;
                borde = BordeCampo;
                break;

            default:
                fondo = Verde;
                hover = VerdeHover;
                activo = VerdeActivo;
                texto = Color.White;
                borde = Verde;
                break;
        }

        EstadoBoton estado = EstadosBotones.GetValue(
            boton,
            _ =>
            {
                var nuevo = new EstadoBoton();

                boton.Paint += PintarBoton;

                boton.MouseEnter += (_, _) =>
                {
                    nuevo.EstaEncima = true;
                    boton.Invalidate();
                };

                boton.MouseLeave += (_, _) =>
                {
                    nuevo.EstaEncima = false;
                    nuevo.EstaPresionado = false;
                    boton.Invalidate();
                };

                boton.MouseDown += (_, e) =>
                {
                    if (e.Button == MouseButtons.Left)
                    {
                        nuevo.EstaPresionado = true;
                        boton.Invalidate();
                    }
                };

                boton.MouseUp += (_, _) =>
                {
                    nuevo.EstaPresionado = false;
                    boton.Invalidate();
                };

                boton.Resize += (_, _) => AplicarRegionBoton(boton);

                return nuevo;
            });

        estado.Fondo = fondo;
        estado.Hover = hover;
        estado.Activo = activo;
        estado.Texto = texto;
        estado.Borde = borde;

        boton.FlatStyle = FlatStyle.Flat;
        boton.FlatAppearance.BorderSize = 0;
        boton.UseVisualStyleBackColor = false;

        boton.BackColor = fondo;
        boton.ForeColor = texto;
        boton.Font = FuenteBoton;
        boton.Cursor = Cursors.Hand;

        boton.Height = AlturaBoton;
        boton.Width = AnchoBoton;
        boton.TextAlign = ContentAlignment.MiddleCenter;

        boton.FlatAppearance.MouseOverBackColor = hover;
        boton.FlatAppearance.MouseDownBackColor = activo;

        AplicarRegionBoton(boton);
        boton.Invalidate();
    }

    private static void PintarBoton(object? sender, PaintEventArgs e)
    {
        if (sender is not Button boton ||
            !EstadosBotones.TryGetValue(boton, out EstadoBoton? estado))
            return;

        if (boton.Width <= 0 || boton.Height <= 0)
            return;

        Graphics g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;
        g.Clear(boton.Parent?.BackColor ?? Fondo);

        Rectangle rect = new(
            1,
            1,
            boton.Width - 3,
            boton.Height - 3);

        Color colorSuperior = estado.Fondo;
        Color colorInferior = estado.Fondo;

        if (estado.EstaPresionado)
        {
            colorSuperior = estado.Activo;
            colorInferior = estado.Activo;
        }
        else if (estado.EstaEncima)
        {
            colorSuperior = estado.Hover;
            colorInferior = estado.Fondo;
        }

        using GraphicsPath ruta = CrearRutaRedondeada(rect, RadioBoton);

        using LinearGradientBrush degradado = new(
            rect,
            colorSuperior,
            colorInferior,
            LinearGradientMode.Vertical);

        g.FillPath(degradado, ruta);

        using Pen lapiz = new(estado.Borde, 1F);
        g.DrawPath(lapiz, ruta);

        TextRenderer.DrawText(
            g,
            boton.Text,
            boton.Font,
            rect,
            estado.Texto,
            TextFormatFlags.HorizontalCenter |
            TextFormatFlags.VerticalCenter |
            TextFormatFlags.EndEllipsis);
    }

    private static void AplicarRegionBoton(Button boton)
    {
        if (boton.Width <= 0 || boton.Height <= 0)
            return;

        using GraphicsPath ruta = CrearRutaRedondeada(
            new Rectangle(0, 0, boton.Width, boton.Height),
            RadioBoton);

        boton.Region?.Dispose();
        boton.Region = new Region(ruta);
    }

    public static void EstilizarTextBox(TextBox caja)
    {
        caja.BorderStyle = BorderStyle.FixedSingle;
        caja.BackColor = FondoCampo;
        caja.ForeColor = Texto;
        caja.Font = FuenteCampo;

        caja.Margin = new Padding(3, 5, 3, 5);

        if (!caja.Multiline)
            caja.Height = AlturaCampo;
    }

    public static void EstilizarComboBox(ComboBox combo)
    {
        combo.FlatStyle = FlatStyle.Flat;
        combo.BackColor = FondoCampo;
        combo.ForeColor = Texto;
        combo.Font = FuenteCampo;
        combo.Cursor = Cursors.Hand;
        combo.Height = AlturaCampo;
    }

    public static void EstilizarFecha(DateTimePicker fecha)
    {
        fecha.Font = FuenteCampo;
        fecha.Format = DateTimePickerFormat.Custom;
        fecha.CustomFormat = "dd/MM/yyyy";

        fecha.CalendarMonthBackground = Blanco;
        fecha.CalendarTitleBackColor = VerdeOscuro;
        fecha.CalendarTitleForeColor = Blanco;
        fecha.CalendarForeColor = Texto;
    }

    public static void EstilizarTarjeta(Panel panel)
    {
        panel.BackColor = Blanco;

        panel.Paint -= PintarTarjeta;
        panel.Paint += PintarTarjeta;

        panel.Resize -= ActualizarRegionTarjeta;
        panel.Resize += ActualizarRegionTarjeta;

        ActualizarRegionTarjeta(panel, EventArgs.Empty);
        panel.Invalidate();
    }

    private static void PintarTarjeta(object? sender, PaintEventArgs e)
    {
        if (sender is not Panel panel ||
            panel.Width <= 0 ||
            panel.Height <= 0)
            return;

        Graphics g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;

        Rectangle rect = new(
            1,
            1,
            panel.Width - 3,
            panel.Height - 3);

        using GraphicsPath ruta = CrearRutaRedondeada(
            rect,
            RadioTarjeta);

        using SolidBrush fondo = new(Blanco);
        using Pen borde = new(Borde, 1F);

        g.FillPath(fondo, ruta);
        g.DrawPath(borde, ruta);
    }

    private static void ActualizarRegionTarjeta(
        object? sender,
        EventArgs e)
    {
        if (sender is not Panel panel ||
            panel.Width <= 0 ||
            panel.Height <= 0)
            return;

        using GraphicsPath ruta = CrearRutaRedondeada(
            new Rectangle(0, 0, panel.Width, panel.Height),
            RadioTarjeta);

        panel.Region?.Dispose();
        panel.Region = new Region(ruta);
    }

        tabla.CellBorderStyle =

        tabla.RowHeadersVisible = false;

        tabla.AllowUserToAddRows = false;
        tabla.AllowUserToDeleteRows = false;
        tabla.AllowUserToResizeRows = false;

        tabla.MultiSelect = false;
        tabla.ReadOnly = true;

        tabla.SelectionMode =
            DataGridViewSelectionMode.FullRowSelect;

        tabla.AutoSizeColumnsMode =
            DataGridViewAutoSizeColumnsMode.Fill;

        tabla.ColumnHeadersBorderStyle =
            DataGridViewHeaderBorderStyle.None;

        tabla.ColumnHeadersHeightSizeMode =
            DataGridViewColumnHeadersHeightSizeMode.DisableResizing;

        tabla.ColumnHeadersHeight = AlturaCabeceraTabla;

        tabla.RowTemplate.Height = AlturaFila;

        tabla.DefaultCellStyle = new DataGridViewCellStyle

            BackColor = Blanco,
            ForeColor = Texto,
            Font = Fuente,
            SelectionBackColor = VerdeClaro,
            SelectionForeColor = Texto,
            Padding = new Padding(10, 0, 0, 0)

        tabla.ColumnHeadersDefaultCellStyle =
            new DataGridViewCellStyle
            {
                BackColor = VerdeOscuro,
                ForeColor = Blanco,
                Font = FuenteNegrita,
                SelectionBackColor = VerdeOscuro,
                SelectionForeColor = Blanco,
                Alignment = DataGridViewContentAlignment.MiddleLeft,
                Padding = new Padding(10, 0, 0, 0)

        tabla.AlternatingRowsDefaultCellStyle =
            new DataGridViewCellStyle

                BackColor = FilaAlterna,
                ForeColor = Texto,
                SelectionBackColor = VerdeClaro,
                SelectionForeColor = Texto,
                Font = Fuente
            };
    }

    // =========================================================
    // APLICAR TEMA A TODO EL FORMULARIO
    // =========================================================

    public static void AplicarTema(Control raiz)
    {
        if (raiz is Form formulario)
        {
            formulario.BackColor = Fondo;
            formulario.Font = Fuente;
        }

        foreach (Control control in raiz.Controls)
        {
            if (control is Button boton)
            {
                EstilizarBoton(
                    boton,
                    ClasificarBoton(boton.Text));
            }
            else if (control is TextBox caja)
            {
                EstilizarTextBox(caja);
            }
            else if (control is ComboBox combo)
            {
                EstilizarComboBox(combo);
            }
            else if (control is DateTimePicker fecha)
            {
                EstilizarFecha(fecha);
            }
            else if (control is DataGridView tabla)
            {
                EstilizarTabla(tabla);
            }
            else if (control is GroupBox grupo)
            {
                EstilizarGrupo(grupo);
            }
            else if (control is Label etiqueta)
            {
                EstilizarEtiqueta(etiqueta);
            }

            if (control.HasChildren)
                AplicarTema(control);
        }
    }

    // =========================================================
    // CLASIFICACIÓN AUTOMÁTICA DE BOTONES
    // =========================================================

    private static TipoBoton ClasificarBoton(string texto)
    {
        string t = (texto ?? string.Empty)
            .Trim()
            .ToLowerInvariant();

        if (t.Contains("eliminar") ||
            t.Contains("borrar") ||
            t.Contains("quitar"))
        {
            return TipoBoton.Peligro;
        }

        if (t.Contains("limpiar") ||
            t.Contains("cancelar") ||
            t.Contains("cerrar") ||
            t.Contains("actualizar") ||
            t.Contains("volver"))
        {
            return TipoBoton.Secundario;
        }

        return TipoBoton.Primario;
    }

    // =========================================================
    // MÉTODO PARA DIBUJAR ESQUINAS REDONDEADAS
    // =========================================================

    private static GraphicsPath CrearRutaRedondeada(
        Rectangle rectangulo,
        int radio)
    {
        GraphicsPath ruta = new();

        int diametro = radio * 2;

        if (diametro > rectangulo.Width)
            diametro = rectangulo.Width;

        if (diametro > rectangulo.Height)
            diametro = rectangulo.Height;

        if (diametro <= 0)
        {
            ruta.AddRectangle(rectangulo);
            return ruta;
        }

        ruta.AddArc(
            rectangulo.X,
            rectangulo.Y,
            diametro,
            diametro,
            180,
            90);

        ruta.AddArc(
            rectangulo.Right - diametro,
            rectangulo.Y,
            diametro,
            diametro,
            270,
            90);

        ruta.AddArc(
            rectangulo.Right - diametro,
            rectangulo.Bottom - diametro,
            diametro,
            diametro,
            0,
            90);

        ruta.AddArc(
            rectangulo.X,
            rectangulo.Bottom - diametro,
            diametro,
            diametro,
            90,
            90);

        ruta.CloseFigure();

        return ruta;
    }
}

    {

        etiqueta.ForeColor = Color.White;
        etiqueta.BackColor = Color.Transparent;
        etiqueta.AutoSize = true;
    }

    {
        etiqueta.Font = TemaSIGEPOSTA.Subtitulo;
        etiqueta.ForeColor = Color.FromArgb(200, 225, 210);
        etiqueta.BackColor = Color.Transparent;
        etiqueta.AutoSize = true;
    }

    // Banner de módulo: fondo verde oscuro con título y subtítulo en blanco.
    public static void AplicarBanner(Panel banner, Label titulo, Label subtitulo = null)
    {
        banner.BackColor = TemaSIGEPOSTA.VerdeOscuro;
        banner.BorderStyle = BorderStyle.None;
        banner.Padding = new Padding(24, 0, 24, 0);

        AplicarTituloBanner(titulo);
        titulo.Location = new Point(24, subtitulo == null ? (banner.Height - titulo.Height) / 2 : 14);

        if (subtitulo != null)
        {
            AplicarSubtituloBanner(subtitulo);
            subtitulo.Location = new Point(26, titulo.Bottom + 2);
        }
    }

    // Tarjeta blanca con borde fino (para agrupar campos o tablas).
    public static void AplicarTarjeta(Panel panel)
    {
        panel.BackColor = Color.White;
        panel.BorderStyle = BorderStyle.None;
        panel.Padding = new Padding(16);

        panel.Paint -= PintarBordeTarjeta;
        panel.Paint += PintarBordeTarjeta;
        panel.Invalidate();
    }

    private static void PintarBordeTarjeta(object sender, PaintEventArgs e)
    {
        Control c = (Control)sender;

        using (Pen lapiz = new Pen(TemaSIGEPOSTA.Borde, 1f))
        {
            e.Graphics.DrawRectangle(lapiz, 0, 0, c.Width - 1, c.Height - 1);
        }
    }

    // Título de sección dentro de una tarjeta (ej. "Datos del paciente").

    {
        etiqueta.Font = new Font("Segoe UI Semibold", 11.5F);
        etiqueta.ForeColor = TemaSIGEPOSTA.VerdeOscuro;
        etiqueta.AutoSize = true;

    {
        TemaSIGEPOSTA.EstilizarEtiqueta(etiqueta);
        etiqueta.AutoSize = true;
    }

    public static void AplicarCompleto(Form formulario)
    {
        Aplicar(formulario);

    }

using System.Drawing;
using System.Windows.Forms;

namespace SIGEPOSTA.Presentacion.Estilos;

public static class EstiloFormulario
{
    public static void Aplicar(Form formulario)
    {
        formulario.BackColor = TemaSIGEPOSTA.Fondo;
        formulario.Font = TemaSIGEPOSTA.Fuente;
        formulario.ForeColor = TemaSIGEPOSTA.Texto;

        AplicarControles(formulario.Controls);
    }

    private static void AplicarControles(Control.ControlCollection controles)
    {
        foreach (Control control in controles)
        {
            // Etiquetas
            if (control is Label etiqueta)
            {
                etiqueta.ForeColor = TemaSIGEPOSTA.Texto;
                etiqueta.Font = TemaSIGEPOSTA.Fuente;
            }

            // Cuadros de texto
            else if (control is TextBox texto)
            {
                texto.Font = TemaSIGEPOSTA.Fuente;
                texto.BackColor = Color.White;
                texto.ForeColor = TemaSIGEPOSTA.Texto;
            }

            // Listas desplegables
            else if (control is ComboBox combo)
            {
                combo.Font = TemaSIGEPOSTA.Fuente;
                combo.BackColor = Color.White;
                combo.ForeColor = TemaSIGEPOSTA.Texto;
                combo.FlatStyle = FlatStyle.Flat;
            }

            // Botones
            else if (control is Button boton)
            {
                boton.Font = new Font("Segoe UI", 9.5F, FontStyle.Bold);
                boton.FlatStyle = FlatStyle.Flat;
                boton.FlatAppearance.BorderSize = 0;
                boton.BackColor = TemaSIGEPOSTA.VerdeOscuro;
                boton.ForeColor = Color.White;
                boton.Cursor = Cursors.Hand;
            }

            // Aplicar también a los controles internos
            if (control.HasChildren)
            {
                AplicarControles(control.Controls);
            }
        }
    }
}

    public static void AplicarPeligro(Button boton)
    {
        boton.UseVisualStyleBackColor = false;
        boton.BackColor = TemaSIGEPOSTA.Rojo;
        boton.ForeColor = Color.White;
        boton.Font = TemaSIGEPOSTA.FuenteBoton;
        boton.FlatStyle = FlatStyle.Flat;
        boton.FlatAppearance.BorderSize = 1;
        boton.FlatAppearance.BorderColor = TemaSIGEPOSTA.Rojo;
        boton.FlatAppearance.MouseOverBackColor = TemaSIGEPOSTA.RojoHover;
        boton.FlatAppearance.MouseDownBackColor = TemaSIGEPOSTA.RojoActivo;
        boton.TextAlign = ContentAlignment.MiddleCenter;
        boton.Cursor = Cursors.Hand;
        boton.Height = TemaSIGEPOSTA.AlturaBoton;
    }

    // ------------- Nuevos -------------

    // Botón del menú lateral que corresponde a la pantalla actual.
    public static void AplicarMenuActivo(Button boton)
    {
        AplicarMenu(boton);
        boton.BackColor = TemaSIGEPOSTA.Verde;
        boton.Font = TemaSIGEPOSTA.FuenteNegrita;
        boton.FlatAppearance.MouseOverBackColor = TemaSIGEPOSTA.Verde;
    }

    // Quita el estado activo (vuelve al estilo normal del menú).
    public static void QuitarMenuActivo(Button boton)
    {
        AplicarMenu(boton);
    }

    // Mismo ancho para un grupo de botones (Registrar, Actualizar, Eliminar, Limpiar).
    public static void IgualarAncho(int ancho, params Button[] botones)
    {
        foreach (Button b in botones)
        {
            b.Width = ancho;
        }
    }
